/* ==========================================================
   أدوات مساعدة

const $ = s => document.querySelector(s);
const scenes = document.querySelectorAll('.scene');
const REDUCED = matchMedia('(prefers-reduced-motion: reduce)').matches;

// الانتقال بين المشاهد بنعومة
function show(id) {
  scenes.forEach(s => s.classList.toggle('active', s.id === id));
  if (id === 'main' || id === 'finale') $('#' + id).scrollTo({ top: 0 });
}

/* ==========================================================
   الجزيئات: ذهب + بتلات + قلوب (خفيفة على الموبايل)
   ========================================================== */
const cv = $('#fx'), cx = cv.getContext('2d');
let W, H, parts = [];
function resize() {
  const d = Math.min(devicePixelRatio || 1, 2);
  W = innerWidth; H = innerHeight;
  cv.width = W * d; cv.height = H * d;
  cx.setTransform(d, 0, 0, d, 0, 0);
}
addEventListener('resize', resize); resize();

const GOLDS = ['#d6b46e', '#ecd49b', '#f6e7bf'];
const PETALS = ['#c99aa0', '#d9aeb3', '#8e6a86', '#e6c3c6'];

function spawn(kind, x, y, burst) {
  const a = Math.random() * Math.PI * 2, sp = burst ? 3 + Math.random() * 7 : 0;
  parts.push({
    kind, x: x ?? Math.random() * W, y: y ?? -20,
    vx: burst ? Math.cos(a) * sp : (Math.random() - .5) * .6,
    vy: burst ? Math.sin(a) * sp - 3 : .6 + Math.random() * 1.1,
    g: burst ? .12 : 0, s: 4 + Math.random() * 8, r: Math.random() * 6, vr: (Math.random() - .5) * .08,
    life: burst ? 150 + Math.random() * 60 : 9999,
    c: kind === 'petal' ? PETALS[Math.random() * 4 | 0] : GOLDS[Math.random() * 3 | 0]
  });
}
function draw(p) {
  cx.save(); cx.translate(p.x, p.y); cx.rotate(p.r); cx.fillStyle = p.c;
  cx.globalAlpha = Math.min(1, p.life / 40);
  if (p.kind === 'heart') { cx.font = p.s * 2 + 'px serif'; cx.fillText('♥', -p.s, p.s); }
  else if (p.kind === 'star') { cx.font = p.s * 2 + 'px serif'; cx.fillText('✦', -p.s, p.s); }
  else if (p.kind === 'petal') { cx.beginPath(); cx.ellipse(0, 0, p.s, p.s / 2, 0, 0, 7); cx.fill(); }
  else { cx.fillRect(-p.s / 2, -p.s / 4, p.s, p.s / 2); }
  cx.restore();
}
function loop() {
  cx.clearRect(0, 0, W, H);
  parts = parts.filter(p => p.life > 0 && p.y < H + 40);
  for (const p of parts) {
    p.vy += p.g; p.x += p.vx + Math.sin(p.y / 40) * .3; p.y += p.vy; p.r += p.vr; p.life--;
    draw(p);
  }
  requestAnimationFrame(loop);
}
loop();

// جزيئات هادية مستمرة (بحد أقصى عشان الأداء)
let ambient = 'dust';
setInterval(() => {
  if (REDUCED || document.hidden || parts.length > 45) return;
  const k = ambient === 'main' ? (Math.random() < .5 ? 'petal' : 'heart') : (Math.random() < .6 ? 'star' : 'petal');
  spawn(k);
}, 650);

// كونفيتي أنيق (عدد محسوب)
function confetti(x = W / 2, y = H / 2, n = 70) {
  if (REDUCED) n = 20;
  const kinds = ['rect', 'rect', 'star', 'heart', 'petal'];
  for (let i = 0; i < n; i++) spawn(kinds[Math.random() * 5 | 0], x, y, true);
}

/* ==========================================================
   العد التنازلي
   ========================================================== */
const b = CONFIG.birthday;
const target = new Date(b.year, b.month, b.day, 0, 0, 0);
const endDay = new Date(b.year, b.month, b.day + 1, 0, 0, 0);
const cd = $('#countdown'), cm = $('#countMsg');

function tick() {
  const now = new Date();
  if (now < target) {
    const t = target - now, p = n => String(n).padStart(2, '0');
    const parts = [[Math.floor(t / 864e5), 'يوم'], [Math.floor(t / 36e5) % 24, 'ساعة'],
                   [Math.floor(t / 6e4) % 60, 'دقيقة'], [Math.floor(t / 1e3) % 60, 'ثانية']];
    cd.innerHTML = parts.map(([v, l]) => `<div><b>${p(v)}</b><span>${l}</span></div>`).join('');
    cm.textContent = 'باقي على يومك الحلو...';
  } else if (now < endDay) {
    cd.innerHTML = ''; cm.textContent = CONFIG.msgToday;
    $('#heroTitle').textContent = CONFIG.heroToday;
  } else {
    cd.innerHTML = ''; cm.textContent = CONFIG.msgAfter;   // بعد العيد: مفيش عداد سالب
  }
}
tick(); setInterval(tick, 1000);

  const ph = el.querySelector('.ph');
  const img = new Image();               // لو الصورة موجودة بتتحمّل، لو لأ يفضل الإطار الوردي
  img.onload = () => { ph.style.backgroundImage = `url("${p.src}")`; ph.textContent = ''; el.dataset.ok = 1; };
  img.src = p.src;
  el.onclick = () => {
    if (!el.dataset.ok) return;
    lb.querySelector('img').src = p.src; lb.querySelector('p').textContent = p.caption;
    lb.classList.add('show');
  };
  tl.appendChild(el);
});

/* ==========================================================
   Reveal عند الظهور
   ========================================================== */
const io = new IntersectionObserver(es => es.forEach(e => {
  if (e.isIntersecting) { e.target.classList.add('in'); io.unobserve(e.target); }
}), { root: null, threshold: .15 });
document.querySelectorAll('.reveal').forEach(el => io.observe(el));
// الـ main عنصر ثابت بيعمل scroll جواه، فبنراقب عن طريق scroll event كمان
$('#main').addEventListener('scroll', () => {
  document.querySelectorAll('.reveal:not(.in)').forEach(el => {
    if (el.getBoundingClientRect().top < innerHeight * .9) el.classList.add('in');
  });
}, { passive: true });
setTimeout(() => $('#main').dispatchEvent(new Event('scroll')), 6000);
